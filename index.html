// File: projects-code/ico-erp-demo/backend/SistemasController.java
@RestController
@RequestMapping("/api/sistemas")
public class SistemasController {

    @Autowired
    private ActivoTiRepository activoTiRepository;

    @PostMapping("/activos-ti")
    @Transactional
    public ResponseEntity<?> crearActivoTi(@RequestBody ActivoTi activo) {
        try {
            if (activo.getCodigoPatrimonial() == null) {
                String prefix = "ICO-LAP-";
                long count = activoTiRepository.count() + 1;
                activo.setCodigoPatrimonial(prefix + String.format("%04d", count));
            }
            ActivoTi guardado = activoTiRepository.save(activo);
            return ResponseEntity.ok(guardado);
        } catch (Exception ex) {
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(ex.getMessage());
        }
    }
}

@Component({
  selector: 'app-sistemas-activos',
  standalone: true,
  templateUrl: './activos.html'
})
export class SistemasActivos implements OnInit {
  private fb = inject(FormBuilder);
  private sistemasService = inject(SistemasService);

  initFormActivo() {
    this.formActivo = this.fb.group({
      codigoPatrimonial: [''],
      tipoActivo: ['LAPTOP', Validators.required],
      marca: ['', Validators.required]
    });

    this.formActivo.get('tipoActivo')?.valueChanges.subscribe(tipo => {
      const nextCode = this.generarSiguienteCodigo(tipo);
      this.formActivo.patchValue({ codigoPatrimonial: nextCode }, { emitEvent: false });
    });
  }
}
